/*
 * ==============================================================================
 * DUAL-MODE SOLAR CLEANER: SPOT (BIRD DROPPINGS) + FULL RASTER (DIFFUSE DUST)
 * ==============================================================================
 * 
 * LOGIC MODES:
 * 1. SPOT CLEANING (Bird waste / localized patch):
 *    - Dual X-motors drive vertical gantry directly to target column.
 *    - Y-motor drives cleaner carriage directly down to target spot.
 *    - Mist nozzle & rotary brush scrub the spot clean (saves 84% energy/water).
 * 
 * 2. FULL RASTER SWEEP (Diffuse dust all over):
 *    - Gantry scans across all columns from Left to Right.
 *    - Carriage sweeps vertically Up and Down each column with brush and spray.
 *    - 100% of panel area is washed cleanly in a serpentine path.
 */

#include <Wire.h>
#include <Adafruit_INA219.h>

#define PIN_CAM_IN     13  // Interrupt signal from ESP32-CAM
#define PIN_X1_STEP    16  // Top horizontal motor
#define PIN_X1_DIR     17
#define PIN_X2_STEP    26  // Bottom horizontal motor (synchronized)
#define PIN_X2_DIR     27
#define PIN_Y_STEP     18  // Vertical carriage motor
#define PIN_Y_DIR      19
#define PIN_BRUSH      25  // Rotary microfiber roller
#define PIN_PUMP        4  // 12V mist spray pump

Adafruit_INA219 ina219;

// Drive dual X-motors at synchronized pulse rate
void stepDualX(int steps, bool moveRight) {
  digitalWrite(PIN_X1_DIR, moveRight ? HIGH : LOW);
  digitalWrite(PIN_X2_DIR, moveRight ? HIGH : LOW);
  for (int s = 0; s < steps; s++) {
    digitalWrite(PIN_X1_STEP, HIGH);
    digitalWrite(PIN_X2_STEP, HIGH);
    delayMicroseconds(600);
    digitalWrite(PIN_X1_STEP, LOW);
    digitalWrite(PIN_X2_STEP, LOW);
    delayMicroseconds(600);
  }
}

// Drive vertical carriage Up or Down
void stepCarriageY(int steps, bool moveDown) {
  digitalWrite(PIN_Y_DIR, moveDown ? HIGH : LOW);
  for (int s = 0; s < steps; s++) {
    digitalWrite(PIN_Y_STEP, HIGH);
    delayMicroseconds(600);
    digitalWrite(PIN_Y_STEP, LOW);
    delayMicroseconds(600);
  }
}

// Full Serpentine Raster Sweep across entire panel (For Diffuse Dust)
void performFullPanelRasterClean() {
  Serial.println(F("[GANTRY] Initiating FULL-PANEL Serpentine Raster Wash..."));
  digitalWrite(PIN_PUMP, HIGH);
  digitalWrite(PIN_BRUSH, HIGH);

  int columns = 6;
  int stepsPerCol = 640;
  int fullYStroke = 2400;

  for (int c = 0; c < columns; c++) {
    bool moveDown = (c % 2 == 0);
    Serial.printf("  >> Column %d/%d: Vertical sweep (%s)...\n", c + 1, columns, moveDown ? "DOWN" : "UP");
    stepCarriageY(fullYStroke, moveDown);
    delay(100);

    // Step gantry right to next column
    if (c < columns - 1) {
      stepDualX(stepsPerCol, true);
      delay(100);
    }
  }

  // Turn off spray and brush
  digitalWrite(PIN_PUMP, LOW);
  digitalWrite(PIN_BRUSH, LOW);

  // Return to Home (Park)
  Serial.println(F("[GANTRY] Returning gantry to Home (Left, Top)..."));
  stepDualX((columns - 1) * stepsPerCol, false); // Return left
  stepCarriageY(fullYStroke, false);             // Return top
  Serial.println(F("[GANTRY] Full clean complete. Gantry securely parked."));
}

// Targeted Spot Clean (For Isolated Bird Waste / Dust Patch)
void performTargetedSpotClean(int targetCol, int targetRow) {
  Serial.printf("[GANTRY] Targeted Spot Wash: Moving directly to Col %d, Row %d...\n", targetCol, targetRow);
  
  // Move directly to spot coordinates
  stepDualX(targetCol * 640, true);
  stepCarriageY(targetRow * 480, true);

  // Scrub and spray localized spot
  digitalWrite(PIN_PUMP, HIGH);
  digitalWrite(PIN_BRUSH, HIGH);
  delay(1800);
  digitalWrite(PIN_PUMP, LOW);
  digitalWrite(PIN_BRUSH, LOW);

  // Return directly to Home
  stepDualX(targetCol * 640, false);
  stepCarriageY(targetRow * 480, false);
  Serial.println(F("[GANTRY] Spot scrub complete. Energy saved: 84%."));
}
